function getMVP(t){
  // Slow cinematic camera orbit after the scene forms.
  const eff=Math.max(0,t)
  const yaw=eff>3.2 ? Math.sin((eff-3.2)*.16)*.20 : 0
  const eye=[Math.sin(yaw)*7.95,3.55,Math.cos(yaw)*7.95]
  const target=[0,1.08,0]
  const view=m4LookAt(eye,target,[0,1,0])
  const proj=m4Perspective(42*Math.PI/180,WG.w/WG.h,.1,40)
  return m4Mul(proj,view)
}
function drawSet(name,mvp,mix,alpha,scale,time=0,mode=0){
  const gl=WG.gl
  gl.useProgram(WG.program)
  bindParticleBuffer(name)
  gl.uniformMatrix4fv(WG.loc.uMVP,false,mvp)
  gl.uniform1f(WG.loc.uMix,mix)
  gl.uniform1f(WG.loc.uAlpha,alpha)
  gl.uniform1f(WG.loc.uPointScale,scale*WG.dpr)
  gl.uniform1f(WG.loc.uTime,time)
  gl.uniform1f(WG.loc.uMode,mode)
  gl.drawArrays(gl.POINTS,0,WG.counts[name])
}


function drawFx(total,t){
  const c=WG.fctx
  const w=WG.w,h=WG.h,cx=w*.5,cy=h*.61
  c.clearRect(0,0,w,h)

  // Dense, clearly visible particle veil across the whole screen.
  // This is intentionally much denser than the previous version.
  c.save()
  c.globalCompositeOperation='screen'
  for(const s of WG.bgScreen){
    const tw=.45+.55*(.5+.5*Math.sin(total*.001*s.s+s.p))
    const alpha=s.a*tw
    c.fillStyle=s.pink
      ?`rgba(255,125,185,${alpha})`
      :`rgba(255,245,250,${alpha*.82})`
    c.beginPath()
    c.arc(s.x,s.y,s.r*(.78+tw*.42),0,Math.PI*2)
    c.fill()
  }
  c.restore()

  // entry burst / fly-in tunnel
  if(t>=0 && t<1.18){
    const p=wClamp(t/1.0,0,1)
    c.save()
    c.globalCompositeOperation='screen'
    for(let i=0;i<92;i++){
      const a=i/92*Math.PI*2 + total*.0009
      const rr=(18 + Math.pow(p,1.6)*(Math.min(w,h)*(.34 + (i%7)*.010)))
      const rr2=Math.max(6,rr-(24+p*82))
      const x1=cx+Math.cos(a)*rr2
      const y1=cy*.88+Math.sin(a)*rr2*.62
      const x2=cx+Math.cos(a)*rr
      const y2=cy*.88+Math.sin(a)*rr*.62
      c.strokeStyle=i%6===0?`rgba(255,255,255,${(1-p)*.34})`:`rgba(255,90,160,${(1-p)*.24})`
      c.lineWidth=1.0+(i%4)*.20
      c.beginPath();c.moveTo(x1,y1);c.lineTo(x2,y2);c.stroke()
    }
    c.restore()
  }

  // soft galaxy haze and bloom
  const gp=wClamp(t/1.35,0,1)
  if(gp>0){    // central glow
    let g=c.createRadialGradient(cx,cy,0,cx,cy,110)
    g.addColorStop(0,'rgba(255,248,252,0.20)')
    g.addColorStop(0.08,'rgba(255,214,231,0.22)')
    g.addColorStop(0.32,'rgba(255,86,164,0.15)')
    g.addColorStop(1,'rgba(255,86,164,0)')
    c.fillStyle=g
    c.beginPath();c.arc(cx,cy,110,0,Math.PI*2);c.fill()
  }

  // heart aura after it rises
  const hp=wClamp((t-1.18)/1.35,0,1)
  if(hp>0.18){
    const hx=cx, hy=cy-122
    const a=(hp-.18)/.82
    let g=c.createRadialGradient(hx,hy,0,hx,hy,86)
    g.addColorStop(0,`rgba(255,228,241,${0.16*a})`)
    g.addColorStop(0.18,`rgba(255,170,206,${0.13*a})`)
    g.addColorStop(0.46,`rgba(255,92,160,${0.09*a})`)
    g.addColorStop(1,'rgba(255,92,160,0)')
    c.fillStyle=g
    c.beginPath();c.arc(hx,hy,86,0,Math.PI*2);c.fill()
  }
}

function projectWorld(p,mvp){
  const c=v4Mul(mvp,[p[0],p[1],p[2],1])
  if(c[3]<=.1)return null
  const nx=c[0]/c[3],ny=c[1]/c[3]
  return {
    x:(nx*.5+.5)*WG.w,
    y:(1-(ny*.5+.5))*WG.h,
    depth:c[3]
  }
}
function updateOverlay(mvp,t,ui){
  const mobile=WG.w<=640;
  const cw=mobile?72:88;
  const ch=mobile?92:112;
  const N=WG.photoItems.length;

  WG.photoItems.forEach((o,i)=>{
    // Evenly-spaced points on ONE spiral path.
    // No random arms -> no bunching together.
    const travel=(o.slot + t*.014) % 1.0;

    // Stay close enough to the heart, but leave enough room between cards.
    const r=1.55 + travel*1.42;
    const a=.35 + travel*Math.PI*4.55;

    const wp=[
      Math.cos(a)*r,
      .19 + .035*Math.sin(a*.75),
      Math.sin(a)*r
    ];

    const p=projectWorld(wp,mvp);
    if(!p)return;

    const scale=wClamp(5.45/p.depth,.78,1.36);

    o.el.style.transform=
      `translate(${p.x-cw/2}px,${p.y-ch/2}px) scale(${scale}) rotate(${Math.cos(a)*4}deg)`;

    // Fade slightly near the wrap endpoints, hiding the reset.
    const edge=Math.min(travel,1-travel);
    const edgeFade=wClamp(edge/.055,0,1);

    o.el.style.opacity=String(
      ui*edgeFade*wClamp(1.54-scale*.18,.56,.97)
    );

    o.el.style.zIndex=String(20+Math.round(90/p.depth));
    o.el.classList.toggle('back',p.depth>9);
  });

  WG.labels.forEach((o,i)=>{
    const travel=(i/WG.labels.length - t*.010 + 1.0) % 1.0;
    const r=1.38 + travel*1.10;
    const a=.75 + travel*Math.PI*4.0;

    const wp=[
      Math.cos(a)*r,
      .10 + .025*Math.sin(a),
      Math.sin(a)*r
    ];

    const p=projectWorld(wp,mvp);
    if(!p)return;

    const w=o.el.offsetWidth||60;
    const h=o.el.offsetHeight||11;
    const scale=wClamp(3.95/p.depth,.56,1.06);

    o.el.style.transform=
      `translate(${p.x-w/2}px,${p.y-h/2}px) scale(${scale}) rotate(${Math.sin(a)*2.4}deg)`;

    o.el.style.opacity=String(ui*.60);
    o.el.style.zIndex='9';
  });
}

function renderWebGL(now){
  if(!WG.active){WG.raf=0;return}
  if(WG.paused){WG.raf=requestAnimationFrame(renderWebGL);return}
  const gl=WG.gl
  const total=(now-WG.start)/1000
  const t=(now-WG.effectStart)/1000
  gl.clearColor(0,0,0,0);gl.clear(gl.COLOR_BUFFER_BIT)

  const mvp=getMVP(t)

  // Background stars always visible.
  drawSet('stars',mvp,1,.72,1.10,Math.max(t,0),3)
  drawFx(now,t)

  if(t>=0){
    // Burst / entry: galaxy points fly from space into their 3D spiral.
    const gm=wClamp(t/1.32,0,1)
    drawSet('galaxy',mvp,gm,1.0,1.22,t,1)

    // Heart rises from the center after the disk forms.
    const hm=wClamp((t-1.18)/1.35,0,1)
    drawSet('heart',mvp,hm,Math.min(1,hm*1.34),1.38,t,2)
    drawSet('heartEdge',mvp,hm,Math.min(1,hm*1.10),1.30,t,0)

    const ui=wClamp((t-2.35)/.85,0,1)
    updateOverlay(mvp,t,ui)
    if(ui>.16&&!WG.scene.classList.contains('ready'))WG.scene.classList.add('ready')
  }

  WG.raf=requestAnimationFrame(renderWebGL)
}

  }
