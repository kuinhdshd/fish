function burst(cx, cy){
  for(let i=0;i<28;i++){
    const s = document.createElement('i'); s.className='spark';
    document.body.appendChild(s);
    const a = Math.random() * Math.PI * 2;
    const d = 45 + Math.random() * 120;
    const size = 3 + Math.random() * 5;
    s.style.width = s.style.height = size + 'px';
    s.style.left = cx + 'px';
    s.style.top = cy + 'px';
    s.animate([
      {transform:'translate(-50%,-50%) scale(1)', opacity:1},
      {transform:`translate(${Math.cos(a)*d}px, ${Math.sin(a)*d}px) scale(.25)`, opacity:0}
    ], {duration:650 + Math.random()*250, easing:'cubic-bezier(.2,.8,.25,1)'}).onfinish = () => s.remove();
  }
}
